add_library(amigaide_core STATIC ${AMIGAIDE_CORE_SOURCES})
target_include_directories(amigaide_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
                                                  ${AMIGAIDE_GENERATED_DIR})
add_dependencies(amigaide_core amigaide_buildstamp)
# So AppSettings can default to the toolchain this build was configured
# against, rather than a hardcoded home directory.
target_compile_definitions(amigaide_core PRIVATE
    AMIGAIDE_DEFAULT_TOOLCHAIN_ROOT="${AMIGAIDE_TOOLCHAIN_ROOT}")
target_link_libraries(amigaide_core PUBLIC Qt6::Widgets Qt6::Concurrent aide_core)
amigaide_set_warnings(amigaide_core)

add_executable(amigaide MACOSX_BUNDLE main.cpp)
target_link_libraries(amigaide PRIVATE amigaide_core)
amigaide_set_warnings(amigaide)

set_target_properties(amigaide PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "AmigaIDE"

    MACOSX_BUNDLE_BUNDLE_VERSION "0.3"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3"

amigaide_add_deploy_target(amigaide)

# Deploy the built binary into the toolchain's bin/ directory, alongside
# vc/vasmm68k_mot/vlink, so it can be launched from the same place as the
# rest of the AmigaOS cross-compile toolchain.
#
# This used to fan out to three destinations, two of them outside the build
# tree entirely (~/Dev/amiga/bin and the source tree's own bin/). One is
# enough: the toolchain root is generated by the bootstrap and is where every
# other tool already lives.
if(APPLE)
    # Only the bundle is deployed. Copying the bare executable out of it
    # produces something that cannot run: macdeployqt rewrites its rpath to
    # @executable_path/../Frameworks, which resolves to <toolchain>/Frameworks
    # once the binary sits in <toolchain>/bin -- the dynamic loader then fails
    # with "Library not loaded: @rpath/QtWidgets.framework". The old tree
    # shipped exactly that broken copy.
    #
    # A launcher script keeps `toolchain/bin/amigaide` working from a shell,
    # and double-clicking the bundle still launches without a Terminal window.
    add_custom_command(TARGET amigaide POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${AMIGAIDE_TOOLCHAIN_ROOT}/bin"
        COMMAND ${CMAKE_COMMAND} -E rm -rf "${AMIGAIDE_TOOLCHAIN_ROOT}/bin/AmigaIDE.app"
        COMMAND ${CMAKE_COMMAND} -E copy_directory $<TARGET_BUNDLE_DIR:amigaide> "${AMIGAIDE_TOOLCHAIN_ROOT}/bin/AmigaIDE.app"
        COMMAND ${CMAKE_COMMAND}
                -DDEST=${AMIGAIDE_TOOLCHAIN_ROOT}/bin/amigaide
                -P "${AMIGAIDE_ROOT}/cmake/WriteBundleLauncher.cmake"
        COMMENT "Deploying AmigaIDE.app into ${AMIGAIDE_TOOLCHAIN_ROOT}/bin"
        VERBATIM
    )
else()
    add_custom_command(TARGET amigaide POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${AMIGAIDE_TOOLCHAIN_ROOT}/bin"
        COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:amigaide> "${AMIGAIDE_TOOLCHAIN_ROOT}/bin/$<TARGET_FILE_NAME:amigaide>"
        COMMENT "Copying amigaide into ${AMIGAIDE_TOOLCHAIN_ROOT}/bin"
        VERBATIM
    )
endif()
